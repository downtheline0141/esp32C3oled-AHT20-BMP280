#include <Arduino.h>
#include <Wire.h>
#include <U8g2lib.h>

#define SDA_PIN 5
#define SCL_PIN 6

// 1. 解像度を 128x64 に変更
// 2. 引数のピン順序は U8g2lib の仕様に合わせて (U8X8_PIN_NONE, SCL, SDA) に指定
// ※アドレスが異なる場合は u8g2_s_0x3c などを試してください
U8G2_SSD1306_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, U8X8_PIN_NONE, SCL_PIN, SDA_PIN);

void setup() {
  // 独自の Wire.begin は省き、U8g2 の内部処理にピンを任せるのが一般的です
  u8g2.begin();
  
  u8g2.setFont(u8g2_font_6x10_tf);
  u8g2.clearBuffer();
  u8g2.drawFrame(28, 24, 70, 39); // 枠のサイズも画面全体（128x64）に変更
  u8g2.drawStr(30, 40, "Hello");
  u8g2.drawStr(30, 50, "ESP32-C3");
  u8g2.sendBuffer();
}

void loop() {}
